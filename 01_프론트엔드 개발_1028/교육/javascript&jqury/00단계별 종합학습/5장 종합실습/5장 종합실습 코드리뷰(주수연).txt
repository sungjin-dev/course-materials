    <script id="practice-code">
        var count = 0;

        initSystem(); //프로그램 시작//

        function initSystem() {  //시스템 시작 함수//
            count++; //1씩 증가//
            document.write("<h3>--- 시스템 접속 횟수: " + count + " ---</h3>"); //h3크기의 '시스템 접속 횟수'라는 글자 출력//

            function User(id, point) { //생성자 함수: 사용자 정보 작성//
                this.userId = id;
                this.userPoint = point;
                this.displayInfo = function() { //작성한 사용자 정보를 출력//
                    return "사용자: " + this.userId + " (현재 포인트: " + this.userPoint + ")";
                };
            }

            var inputId = prompt("아이디를 입력하세요.", "guest"); //입력 받기//
            var newUser = new User(inputId, 1000); //아이디, 1000 저장//
            
            document.write(newUser.displayInfo() + "<br>"); //화면에 출력, 입력 받은 내용(아이디와 1000)//

            var addValue = prompt("추가할 포인트를 입력하세요.", "0");

            if (isNaN(addValue)) { //숫자가 아니면//
                alert("숫자만 입력 가능합니다.");
            } else {
                var pointAmount = parseInt(addValue); //숫자면//
                
                document.write("포인트 적립 시작... ");
                startTimer(3); //타이머 3까지//

                var finalPoint = calculatePoint(newUser.userPoint, pointAmount); //포인트 + 추가 포인트 계산//
                document.write("<br><strong>최종 포인트: " + finalPoint + "</strong>"); //최종 결과 출력//
            }
        }

        function calculatePoint(current, add) { //포인트 계산//
            var processMsg = "계산 중..."; 
            console.log(processMsg);
            return current + add; //값 반환//
        }

        function startTimer(n) { //계산 타이머//
            if (n == 0) {
                document.write(" 완료!");
                return; 
            }
            document.write(n + " "); //재귀 함수//
            startTimer(n - 1);
        }

        function copyCode() { //버튼 클릭하면 실행//
            const code = document.getElementById('practice-code').innerText; //코드 가져오기//
            navigator.clipboard.writeText(code).then(() => { //복사하기//
                alert("코드가 클립보드에 복사되었습니다!"); //복사 성공 알림//
            });
        }
    </script>
