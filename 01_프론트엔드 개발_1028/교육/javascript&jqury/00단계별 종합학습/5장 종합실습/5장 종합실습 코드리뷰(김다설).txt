    <!-- id를 copyBtn로 설정하고 버튼 스타일 적용함, 
         클릭 이벤트가 발생할 때마다 해당 함수를 호출하고 onclick에 copyCode함수의 값을 저장됨 
        버튼 이름은 코드 복사하기--> 
    <button id="copyBtn" onclick="copyCode()">코드 복사하기</button>
    
    <!--javascipt 코드 시작-->
    <script id="practice-code"> // script의 id를 practice-code로 설정
        var count = 0; // count 변수를 선언하고 0으로 초기화

        initSystem(); // initSystem 함수 호출

        // initSystem 함수 생성
        function initSystem() {
            count++; // count = count + 1
            document.write("<h3>--- 시스템 접속 횟수: " + count + " ---</h3>"); // 문자열 + 변수 count + 문자열

            // 객체 User를 만드는 함수 생성 (매개변수 - id, point)
            function User(id, point) {
                this.userId = id; // 객체 User의 userId에는 입력받은 아이디가 저장
                this.userPoint = point; // 객체 User의 userPoint에는 설정한 point 값 1000이 저장
                this.displayInfo = function() {
                    return "사용자: " + this.userId + " (현재 포인트: " + this.userPoint + ")";
                }; //생성된 함수를 통해 만들어진 객체 함수 값을 반환받아 보여줌
            }

            /*new 예약어로 User 객체를 생성하고 생성된 객체에 입력받은 변수 inputId 값과 point 값 1000을 
            newUser라는 변수에 값을 저장*/
            var inputId = prompt("아이디를 입력하세요.", "guest"); // 변수 inputId 생성하고 입력 받음
            var newUser = new User(inputId, 1000); 
            
            /*현재 만들어진 객체 User의 정보를 출력해서 보여줌
              <br>태그로 줄바꿈*/
            document.write(newUser.displayInfo() + "<br>"); 

            /* 변수 addvalue를 선언하고 변수 point에 추가할 값을 입력받아 저장, 기본값은 0으로 설정*/
            var addValue = prompt("추가할 포인트를 입력하세요.", "0"); 
           
            /*addValue 값이 숫자인지 문자인지 유효성 평가*/
            if (isNaN(addValue)) {              // addValue가 숫자가 아닌 경우(참) 실행
                alert("숫자만 입력 가능합니다."); 
            } else {                            // addValue가 숫자인 경우(거짓) 실행
                /* addValue를 10진수 숫자로 변경하여 생성한 변수 pointAmount에 저장 */
                var pointAmount = parseInt(addValue); 
                
                /*문자열 출력함수*/
                document.write("포인트 적립 시작... ");

                /*startTimer 함수에 매개변수 값 3 입력받아 실행*/
                startTimer(3);

                /*startTimer 함수에 매개변수 newUser.userPoint, pointAmount 입력받아 
                  반환된 값을 finalPoint라는 변수에 저장*/
                var finalPoint = calculatePoint(newUser.userPoint, pointAmount);

                 /*문자열 출력함수, 줄바꿈, 강조 태그
                   줄바꿈 + 강조 열린태그 + 문자열 + 변수 finalPoint 값 + 강조 닫힌태그  */
                document.write("<br><strong>최종 포인트: " + finalPoint + "</strong>");
            }
        }

        /* calculatePoint 함수 생성 (매개변수 - current, add)*/
        function calculatePoint(current, add) {
            var processMsg = "계산 중..."; // 문자열을 변수 processMsg에 저장
            console.log(processMsg); // 웹페이지 콘솔창에서 출력되는 내용
            return current + add;   // 매개변수 current에 저장된 point값에 추가한 포인트 값이 더하여 값 반환
        }

        /*startTimer 함수 생성 (매개변수 - n)*/
        function startTimer(n) {
            /* n이 0이면, 문자열 출력하고 함수 종료 */
            if (n == 0) {        
                document.write(" 완료!");
                return; 
            }
            document.write(n + " "); // if 조건이 거짓일 때까지 계속 출력, 변수 n의 값 + " "(공백)
            startTimer(n - 1); // 재귀함수 
        }

        /*copyCode 함수 생성*/
        function copyCode() {
            /* text 안에서 id가 practice-code인 것을 찾아 해당 객체의 주소를 저장*/
            const code = document.getElementById('practice-code').innerText;
            navigator.clipboard.writeText(code).then(() => {
                alert("코드가 클립보드에 복사되었습니다!");
            });
        }
    </script>
